// Map: kharakorum

kharakorum,146,268,4	script	Kafra Employee#khara_base	113,{
    mes "[Kafra Employee]";
    mes "ยินดีต้อนรับค่ะ ต้องการใช้บริการอะไรคะ?";
    next;
    
    // เพิ่มเมนูใหม่เข้าไปในคำสั่ง select
    switch(select("บันทึกจุดเซฟ:เปิดคลังเก็บของ (1,000 Zeny):เปิดคลังกิลด์:เรียกสมาชิกกิลด์มาที่นี่:เรียกสมาชิกปาร์ตี้มาที่นี่:ยกเลิก")) {
        case 1:
            mes "[Kafra Employee]";
            mes "บันทึกจุดเซฟเรียบร้อยแล้วค่ะ ไว้เจอกันใหม่นะคะ";
            // ดึงพิกัดของ NPC ตัวที่ผู้เล่นกำลังคุยอยู่มาทำเป็นจุดเซฟ
            getmapxy(.@map$, .@x, .@y, 1);
            savepoint .@map$, .@x, .@y;
            specialeffect2 17;
            close;
            
        case 2:
            mes "[Kafra Employee]";
            // เช็คว่ามีเงินพอ 1000 zeny หรือไม่
            if (Zeny < 1000) {
                mes "ขออภัยค่ะ คุณมีเงินไม่พอสำหรับค่าบริการ 1,000 Zeny ค่ะ";
                close;
            }
            // หักเงิน 1000 Zeny
            Zeny -= 1000;
            mes "ขอบคุณค่ะ หักค่าบริการ 1,000 Zeny เรียบร้อยแล้วค่ะ";
            close2;
            openstorage;
            end;

        case 3:
            mes "[Kafra Employee]";
            // เช็คว่าผู้เล่นมีกิลด์หรือไม่ (getcharid(2) จะได้ค่า 0 ถ้าไม่มีกิลด์)
            if (getcharid(2) == 0) {
                mes "ขออภัยค่ะ คุณยังไม่ได้สังกัดกิลด์ใดเลยค่ะ";
                close;
            }
            mes "เปิดคลังกิลด์ให้เรียบร้อยแล้วค่ะ";
            close2;
            openguildstorage; // คำสั่งมาตรฐานสำหรับเปิดคลังกิลด์
            end;

        case 4:
            mes "[Kafra Employee]";
            // เช็คว่าผู้เล่นมีกิลด์หรือไม่
            if (getcharid(2) == 0) {
                mes "ขออภัยค่ะ คุณยังไม่ได้สังกัดกิลด์ใดเลยค่ะ";
                close;
            }
            mes "ทำการเรียกสมาชิกกิลด์ทั้งหมดมาที่นี่เรียบร้อยแล้วค่ะ!";
            close2;
            // ดึงพิกัดปัจจุบันของ "ผู้เล่น" (พารามิเตอร์สุดท้ายเป็น 0)
            getmapxy(.@map$, .@x, .@y, 0); 
            // วาร์ปคนในกิลด์มาที่พิกัดที่ดึงมาได้
            warpguild .@map$, .@x, .@y, getcharid(2);
            end;

        case 5:
            mes "[Kafra Employee]";
            // เช็คว่าผู้เล่นมีปาร์ตี้หรือไม่ (getcharid(1) จะได้ค่า 0 ถ้าไม่มีปาร์ตี้)
            if (getcharid(1) == 0) {
                mes "ขออภัยค่ะ คุณยังไม่ได้สร้างหรือเข้าร่วมปาร์ตี้เลยค่ะ";
                close;
            }
            mes "ทำการเรียกสมาชิกปาร์ตี้ทั้งหมดมาที่นี่เรียบร้อยแล้วค่ะ!";
            close2;
            getmapxy(.@map$, .@x, .@y, 0);
            warpparty .@map$, .@x, .@y, getcharid(1); // วาร์ปคนในปาร์ตี้มาที่พิกัดที่ดึงมาได้
            end;
            
        case 6:
            mes "[Kafra Employee]";
            mes "ขอบคุณที่ใช้บริการค่ะ";
            close;
    }
}
