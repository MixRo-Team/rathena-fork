//============================================================
//= Cash Coin Function
//============================================================

function	script	CC_Function	{

	.CashCoin = 30001;
	.Amounts = countitem(.CashCoin);
	delitem .CashCoin,.Amounts;
	.Counts = .Amounts+1;
	#CASHPOINTS += .Amounts+1;
	dispbottom "[ระบบ] : คุณได้รับ "+callfunc("F_InsertComma",.Counts)+" Cash Points | คงเหลือทั้งหมด "+callfunc("F_InsertComma",#CASHPOINTS)+" Cash Points.";
	end;
}

// ==========================================
// ฟังก์ชันเรียกใช้งาน VIP 7 วัน
// ==========================================
function	script	7d_vip_ticket	{
	// 1. เช็คว่ามี VIP อยู่แล้วหรือไม่
	if (#VIP_TIME > gettimetick(2)) {
		dispbottom "[ระบบ VIP] : คุณยังมีสถานะ VIP ใช้งานอยู่ครับ";
		end;
	}

	// 2. บันทึกเวลาหมดอายุ (7 วัน = 604,800 วินาที)
	#VIP_TIME = gettimetick(2) + 604800;

	// 3. รันสถานะ SC_VIP_7D (ที่จารคอมไพล์ C++ ไว้แล้ว)
	sc_start SC_VIP_7D, 604800000, 0;

	specialeffect2 73; // เอฟเฟกต์วิ้งๆ
	dispbottom "^00FF00[ระบบ VIP]^000000 เปิดใช้งาน VIP 7 วันเรียบร้อยแล้วครับ!";
	return;
}
