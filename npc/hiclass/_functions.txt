//============================================================
//= Cash Coin Function
//============================================================

function	script	CC_Function	{

	.CashCoin = 30001;
	.Amounts = countitem(.CashCoin);
	delitem .CashCoin,.Amounts;
	.Counts = .Amounts+1;
	#CASHPOINTS += .Amounts+1;
	dispbottom "[ระบบ] : คุณได้รับ "+callfunc("F_InsertComma",.Counts)+" Cash Points | คงเหลือทั้งหมด "+callfunc("F_InsertComma",#CASHPOINTS)+" Cash Points.";
	end;
}
