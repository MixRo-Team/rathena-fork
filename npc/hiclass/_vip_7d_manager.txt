// ==========================================
// ระบบคุมสถานะ VIP 7 วัน แบบเดี่ยว (Standalone)
// ==========================================
-	script	VIP_Manager	-1,{

OnPCLoginEvent:
OnPCStatCalcEvent:
	// ตรวจสอบว่าเวลา VIP ยังเหลืออยู่ และยังไม่มีบัฟติดตัว
	if (#VIP_TIME > gettimetick(2) && !sc_check(SC_VIP_7D)) {
		// คำนวณเวลาที่เหลือเป็นมิลลิวินาที
		.@time_left = (#VIP_TIME - gettimetick(2)) * 1000;
		sc_start SC_VIP_7D, .@time_left, 0;
	}
	
	// ตรวจสอบกรณีเวลาหมด (ให้ล้างสถานะออก)
	if (#VIP_TIME > 0 && #VIP_TIME <= gettimetick(2)) {
		#VIP_TIME = 0;
		sc_end SC_VIP_7D;
		dispbottom "^FF0000[ระบบ VIP]^000000 สถานะ VIP 7 Day ของคุณหมดอายุแล้วครับ";
	}
	end;
}
