// ==========================================
// ระบบคุมสถานะ VIP อัตโนมัติ (ใส่ไว้ใน scripts_custom.conf)
// ==========================================
-	script	VIP_Manager	-1,{

OnPCLoginEvent:
OnPCStatCalcEvent:
	// ตรวจสอบว่าเวลา VIP ยังเหลืออยู่ และสถานะยังไม่ติดบัฟ
	if (#VIP_TIME > gettimetick(2) && !isstatusactive(SC_VIP_7D)) {
		// คำนวณเวลาที่เหลือเป็นมิลลิวินาที
		.@time_left = (#VIP_TIME - gettimetick(2)) * 1000;
		sc_start SC_VIP_7D, .@time_left, 0;
	}
	
	// ตรวจสอบกรณีเวลาหมด (ล้างสถานะ)
	if (#VIP_TIME > 0 && #VIP_TIME <= gettimetick(2)) {
		#VIP_TIME = 0;
		sc_end SC_VIP_7D;
		dispbottom "[ระบบ VIP] : สถานะ VIP 7 Day ของคุณหมดอายุแล้วครับ";
	}
	end;
}
