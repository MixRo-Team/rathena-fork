// ==========================================
// ตัว NPC หลัก (ตู้ Cash Coin) [รองรับการซื้อทีละหลายชิ้น]
// ==========================================
prontera,144,196,2	script	[ Cash Coin ]	2_VENDING_MACHINE1,{
Start:
	mes "^FF3300[ Cash Coin ]^000000";
	mes "ยินดีต้อนรับสู่ตู้ Cash Coin";
	mes "เลือกหมวดหมู่ไอเทมที่คุณต้องการได้เลยครับ";
	next;
	
	set .@cat, select("- ทั่วไป:- อาวุธตีคน:- อาวุธตีบอส:- หมวก:- ส่วนหู:- ส่วนปาก:- เครื่องประดับ:- ชุดสวมใส่");
	callshop "cash_cat"+.@cat, 1;
	npcshopattach "cash_cat"+.@cat;
	end;

OnBuyItem:
	.@item = @bought_nameid;
	.@qty = @bought_quantity; // ดึงจำนวนที่ผู้เล่นกดซื้อมาเก็บไว้
	.@rate_true = 100;
	.@zeny_use = 0;
	.@cash_use = 0;
	deletearray .@item_quest;
	deletearray .@amount_quest;

	// ==========================================
	// 1. กำหนดราคาพื้นฐาน (ต่อ 1 ชิ้น)
	// ==========================================
	switch(.@item) {
		case 30001: // Cash Coin
			.@cash_use = 1;
			break;

		case 33006: // Bag of Cash Coins 1000
			.@cash_use = 1000;
			break;

		case 5022: // ตัวอย่าง STR Invisible Cap
			.@cash_use = 1000;
			// .@rate_true = 50; 
			break;

		default:
			mes "^FF3300[ Cash Coin ]^000000";
			mes "ไอเทมนี้ยังไม่ได้ตั้งค่าราคาครับ";
			close;
	}

	// ==========================================
	// 2. ระบบคำนวณคูณจำนวนที่ต้องการซื้อ
	// ==========================================
	.@total_cash = .@cash_use * .@qty;
	.@total_zeny = .@zeny_use * .@qty;
	for ( .@i = 0; .@i < getarraysize(.@amount_quest); .@i ++ ) {
		.@amount_quest[.@i] = .@amount_quest[.@i] * .@qty;
	}

	// ==========================================
	// 3. หน้าต่างแสดงผลลัพธ์ (ตามภาพตัวอย่าง)
	// ==========================================
	mes "รายการ ที่เลือกคุณได้เลือก: ^0000FF" + .@qty + "x " + getitemname(.@item) + "^000000";
	mes "รายการที่ คุณต้องมีเพื่อแลกเปลี่ยน:";

	// เช็ก Cash Points
	if (.@total_cash > 0) {
		if (#CASHPOINTS >= .@total_cash) {
			mes "^00FF00• " + callfunc("F_InsertComma", .@total_cash) + " Cash Points (" + callfunc("F_InsertComma", #CASHPOINTS) + "/" + callfunc("F_InsertComma", .@total_cash) + ")^000000";
		} else {
			mes "^FF0000• " + callfunc("F_InsertComma", .@total_cash) + " Cash Points (" + callfunc("F_InsertComma", #CASHPOINTS) + "/" + callfunc("F_InsertComma", .@total_cash) + ")^000000";
		}
	}

	// เช็ก Zeny
	if (.@total_zeny > 0) {
		if (Zeny >= .@total_zeny) {
			mes "^00FF00• " + callfunc("F_InsertComma", .@total_zeny) + " Zeny (" + callfunc("F_InsertComma", Zeny) + "/" + callfunc("F_InsertComma", .@total_zeny) + ")^000000";
		} else {
			mes "^FF0000• " + callfunc("F_InsertComma", .@total_zeny) + " Zeny (" + callfunc("F_InsertComma", Zeny) + "/" + callfunc("F_InsertComma", .@total_zeny) + ")^000000";
		}
	}

	// เช็กไอเทมวัตถุดิบ
	for ( .@i = 0; .@i < getarraysize(.@item_quest); .@i ++ ) {
		.@have = countitem(.@item_quest[.@i]);
		if (.@have >= .@amount_quest[.@i]) {
			mes "^00FF00• " + getitemname(.@item_quest[.@i]) + " (" + callfunc("F_InsertComma", .@have) + "/" + callfunc("F_InsertComma", .@amount_quest[.@i]) + ") ea^000000";
		} else {
			mes "^FF0000• " + getitemname(.@item_quest[.@i]) + " (" + callfunc("F_InsertComma", .@have) + "/" + callfunc("F_InsertComma", .@amount_quest[.@i]) + ") ea^000000";
		}
	}

	if (.@rate_true < 100) {
		mes " ";
		mes "^FF9900โอกาสแลกเปลี่ยนสำเร็จ: " + .@rate_true + "%^000000";
	}

	next;

	// ==========================================
	// 4. เมนู Action และการหักของ
	// ==========================================
	.@type = getiteminfo(.@item, ITEMINFO_TYPE);
	
	if (.@type == 4 || .@type == 5) {
		.@s = select("- แลกเปลี่ยน ^0000FF" + getitemname(.@item) + "^000000:- ทดลองใส่:- ^FF0000ยกเลิก^000000");
	} else {
		.@s = select("- แลกเปลี่ยน ^0000FF" + getitemname(.@item) + "^000000:- ^FF0000ยกเลิก^000000");
		if (.@s == 2) { .@s = 3; }
	}

	switch(.@s) {
		case 1:
			// ป้องกันบั๊กโยนของ
			if (.@total_cash > 0 && #CASHPOINTS < .@total_cash) { mes "^FF3300[ Cash Coin ]^000000"; mes "Cash Points ของคุณไม่เพียงพอครับ!"; close; }
			if (.@total_zeny > 0 && Zeny < .@total_zeny) { mes "^FF3300[ Cash Coin ]^000000"; mes "Zeny ของคุณไม่เพียงพอครับ!"; close; }
			for ( .@i = 0; .@i < getarraysize(.@item_quest); .@i ++ ) {
				if (countitem(.@item_quest[.@i]) < .@amount_quest[.@i]) { mes "^FF3300[ Cash Coin ]^000000"; mes "วัตถุดิบ " + getitemname(.@item_quest[.@i]) + " ของคุณไม่เพียงพอครับ!"; close; }
			}

			// ป้องกันน้ำหนักเกินตามจำนวนที่ซื้อ
			if (checkweight(.@item, .@qty) == 0) {
				mes "^FF3300[ Cash Coin ]^000000";
				mes "น้ำหนัก หรือ ช่องเก็บของของคุณเต็มครับ";
				mes "กรุณาเคลียร์กระเป๋าก่อนครับ";
				close;
			}

			// หักของ (หักตามยอดรวมที่คูณแล้ว)
			if (.@total_cash > 0) { #CASHPOINTS -= .@total_cash; }
			if (.@total_zeny > 0) { Zeny -= .@total_zeny; }
			for ( .@i = 0; .@i < getarraysize(.@item_quest); .@i ++ ) { delitem .@item_quest[.@i], .@amount_quest[.@i]; }

			if (rand(1, 100) > .@rate_true) {
				mes "^FF3300[ Cash Coin ]^000000";
				mes "เสียใจด้วย การแลกเปลี่ยน ^FF0000ล้มเหลว^000000";
				specialeffect2 155;
				close;
			}

			// ให้ไอเทมตามจำนวนที่กดซื้อ
			getitem .@item, .@qty;
			
			// เปลี่ยน Effect จากแสงตีบวก (154) เป็นแสง Level Up สว่างวาบ (73)
			specialeffect2 73; 
			mes "^FF3300[ Cash Coin ]^000000";
			mes "ยินดีด้วย คุณได้รับ ^009900" + .@qty + "x " + getitemname(.@item) + "^000000";
			if (.@rate_true < 100) { announce "["+strnpcinfo(0)+"] : ยินดีด้วย คุณ "+strcharinfo(0)+" ได้รับ "+getitemname(.@item)+" สำเร็จ!", bc_all, 0xFFFF00; }
			close;

		case 2:
			mes "^FF3300[ Cash Coin ]^000000";
			mes "พิมพ์คำสั่งนี้เพื่อดูความสามารถของไอเทม:";
			mes "^0000FF@iteminfo " + .@item + "^000000";
			close;

		case 3:
			close;
	}
	end;

OnInit:
	// กำหนดราคาหลอก (Zeny) ในร้านค้าให้ตรงกับราคา Cash
	npcshopadditem "cash_cat1", 30001, 1;    // แสดงราคา 1 Z (ใช้ 1 Cash)
	npcshopadditem "cash_cat1", 33006, 1000; // แสดงราคา 1000 Z (ใช้ 1000 Cash)
	npcshopadditem "cash_cat4", 5022, 1000;  // แสดงราคา 1000 Z
	end;
}
