// ==========================================
// ตัว NPC หลัก (ตู้ Cash Coin) [รุ่นสมบูรณ์แบบ]
// ==========================================
prontera,144,196,2	script	[ Cash Coin ]	2_VENDING_MACHINE1,{

Start:
	mes "^FF3300[ Cash Coin ]^000000";
	mes "ยินดีต้อนรับสู่ตู้ Cash Coin";
	mes "สามารถแลกเปลี่ยน Cash เป็นเหรียญจำนวนมากได้ที่นี่";
	next;
	
	switch(select("- หมวดหมู่สินค้า (Shop):- แลกเหรียญจำนวนมาก (Mass Exchange)")) {
		case 1:
			set .@cat, select("- ทั่วไป:- อาวุธตีคน:- อาวุธตีบอส:- หมวก:- ส่วนหู:- ส่วนปาก:- เครื่องประดับ:- ชุดสวมใส่");
			callshop "cash_cat"+.@cat, 1;
			npcshopattach "cash_cat"+.@cat;
			end;
		case 2:
			goto OnMassExchange;
			end;
	}

OnMassExchange:
	mes "^FF3300[ Cash Coin ]^000000";
	mes "คุณมีแต้ม: ^009900" + callfunc("F_InsertComma", #CASHPOINTS) + " Cash^000000";
	mes "ต้องการแลก Cash Coin (ID:30001) กี่เหรียญครับ?";
	mes "(อัตรา 1 Cash ต่อ 1 เหรียญ)";
	input .@amount, 1, 10000; // แลกได้ครั้งละ 1 - 10,000 เหรียญ
	
	if (.@amount * 1 > #CASHPOINTS) {
		mes "แต้มของคุณไม่เพียงพอครับ!";
		close;
	}
	
	if (checkweight(30001, .@amount) == 0) {
		mes "น้ำหนักกระเป๋าคุณไม่พอครับ!";
		close;
	}
	
	#CASHPOINTS -= .@amount;
	getitem 30001, .@amount;
	mes "แลกเปลี่ยนสำเร็จ! ได้รับ Cash Coin จำนวน ^009900" + .@amount + " ชิ้น^000000";
	close;

OnBuyItem:
	// [SECURITY] ดัก WPE
	if (@bought_quantity > 1) { mes "^FF3300[ Cash Coin ]^000000"; mes "ระบบป้องกัน: แลกได้ทีละ 1 ชิ้นเท่านั้นครับ"; close; }

	.@item = @bought_nameid;
	.@rate_true = 100;
	.@zeny_use = 0;
	.@cash_use = 0;
	deletearray .@item_quest;
	deletearray .@amount_quest;

	// ==========================================
	// ตารางราคาไอเทม (เพิ่มไอเทมใหม่ได้ที่นี่)
	// ==========================================
	switch(.@item) {
		// หมวดเหรียญ (ถ้าผู้เล่นกดซื้อผ่านหน้าร้าน)
		case 30001: .@cash_use = 1; break;
		case 33006: .@cash_use = 1000; break;
		
		// หมวดหมวกตัวอย่าง
		case 5022: .@cash_use = 1000; break;
		
		default:
			mes "ไอเทมนี้ยังไม่ได้ตั้งค่าราคาครับ";
			close;
	}

	// [โชว์หน้าต่างคอนเฟิร์มเหมือนเดิม]
	mes "รายการ ที่เลือกคุณได้เลือก: ^0000FF" + getitemname(.@item) + "^000000";
	if (.@cash_use > 0) {
		if (#CASHPOINTS >= .@cash_use) { mes "^00FF00• ใช้ " + callfunc("F_InsertComma", .@cash_use) + " Cash (" + #CASHPOINTS + "/" + .@cash_use + ")^000000"; } 
		else { mes "^FF0000• ใช้ " + callfunc("F_InsertComma", .@cash_use) + " Cash (" + #CASHPOINTS + "/" + .@cash_use + ")^000000"; }
	}
	next;

	// เมนูอัจฉริยะ (แยกของสวมใส่/ของทั่วไป)
	.@type = getiteminfo(.@item, ITEMINFO_TYPE);
	if (.@type == 4 || .@type == 5) { .@s = select("- แลกเปลี่ยน " + getitemname(.@item) + ":- ทดลองใส่:- ยกเลิก"); } 
	else { .@s = select("- แลกเปลี่ยน " + getitemname(.@item) + ":- ยกเลิก"); if(.@s==2) .@s=3; }

	switch(.@s) {
		case 1:
			// เช็กของกันบั๊ก
			if (.@cash_use > 0 && #CASHPOINTS < .@cash_use) { close; }
			if (checkweight(.@item, 1) == 0) { mes "น้ำหนักเต็มครับ!"; close; }
			
			// หักเงิน
			#CASHPOINTS -= .@cash_use;
			getitem .@item, 1;
			specialeffect2 154;
			mes "ได้รับ ^009900" + getitemname(.@item) + "^000000 สำเร็จ!";
			close;
		case 2:
			mes "@iteminfo " + .@item;
			close;
	}
	end;

OnInit:
	npcshopadditem "cash_cat1", 30001, -1; // Cash Coin
	npcshopadditem "cash_cat1", 33006, -1; // Bag of Cash Coins
	npcshopadditem "cash_cat4", 5022, -1;  // STR Invisible Cap
	end;
}
