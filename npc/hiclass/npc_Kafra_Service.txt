	// ==========================================
	// 1. หน้าต่างต้อนรับ และแสดงข้อมูลตัวละคร
	// ==========================================
	mes "[ Kafra Corporation ]";
	mes "ยินดีต้อนรับคุณ : ^FF0000" + strcharinfo(0) + "^000000";
	mes "Account Id : ^00AA00" + getcharid(3) + "^000000";
	mes "Char Id : ^00AA00" + getcharid(0) + "^000000";
	mes " "; // เว้นบรรทัดให้สวยงาม

	// ==========================================
	// 2. เมนูเลือกใช้งาน (ตัดโซลิ้งออกแล้ว)
	// ==========================================
	.@menu = select(
		" ~ ^0055FFบันทึกจุดเกิด^000000",
		" ~ ^FF7700เปิดคลังเก็บของ^000000",
		" ~ ^FF7700เปิดคลังกิลด์^000000",
		" ~ ^FF7700ระบบกิลด์^000000",
		" ~ ^990099แต่งตัวละคร^000000",
		" ~ ^008800รีเซ็ต สเตตัส (1m)^000000",
		" ~ ^008800รีเซ็ต สกิล^000000",
		" ~ ^007799เช่าทุกอย่าง^000000",
		" ~ ^007799แก้บัค^000000",
		" ~ ^007799เคลือบธาตุ^000000",
		" ~ ^007799ก็อบปี้สกิล^000000",
		" ~ ^007799เพิ่มน้ำหนัก^000000",
		" ~ ^8B4513สัตว์เลี้ยงสนิททันที^000000",
		" ~ ^8B4513โฮมุนครุส^000000",
		" ~ ^777777เอาไว้คราวหน้า !!^000000"
	);

	switch(.@menu) {
		case 1: // บันทึกจุดเกิด
			getmapxy(.@map$, .@x, .@y, 1); 
			savepoint .@map$, .@x, .@y;
			
			specialeffect 17; 
			mes "ระบบได้ทำการ ^000099บันทึกจุดเกิด^000000 ให้คุณเรียบร้อยแล้วค่ะ";
			mes "ไว้เจอกันใหม่นะคะ";
			close;

		case 2: // เปิดคลังเก็บของ
			close2;
			openstorage;
			end;

		case 3: // เปิดคลังกิลด์
			if (getcharid(2) == 0) {
				mes "^FF0000คุณยังไม่มีกิลด์ค่ะ ไม่สามารถเปิดคลังกิลด์ได้^000000";
				close;
			}
			close2;
			guildopenstorage;
			end;

		case 4: // ระบบกิลด์ (หัก 100k Zeny ถาวร สร้างกิลด์ให้ทันที)
			mes "[ ระบบกิลด์ ]";
			if (getcharid(2) > 0) {
				mes "คุณมีกิลด์อยู่แล้วค่ะ สามารถกด ^0055FF[ Alt+G ]^000000 เพื่อจัดการกิลด์ได้เลย";
				close;
			}
			
			mes "การสร้างกิลด์มีค่าธรรมเนียม ^FF7700100,000 Zeny^000000";
			mes "และ ^FF0000จะไม่สามารถขอคืนเงินได้ในทุกกรณี^000000 ค่ะ";
			mes "คุณต้องการดำเนินการต่อหรือไม่คะ?";
			next;
			
			if (select(" • ตกลง (สร้างกิลด์)", " • ยกเลิก") == 2) {
				close;
			}
			
			if (Zeny < 100000) {
				mes "[ ระบบกิลด์ ]";
				mes "^FF0000Zeny ของคุณไม่พอค่ะ!^000000 ต้องการใช้ 100,000 Zeny ในการสร้างกิลด์";
				close;
			}
			
			Zeny -= 100000;
			
			mes "[ ระบบกิลด์ ]";
			mes "กรุณาใส่ ^0055FFชื่อกิลด์^000000 ที่คุณต้องการสร้างค่ะ:";
			next;
			
			input .@guild_name$;
			
			if (.@guild_name$ == "") {
				mes "คุณไม่ได้ระบุชื่อกิลด์ ระบบได้หักค่าธรรมเนียม ^FF7700100,000 Zeny^000000 เรียบร้อยแล้วค่ะ";
				close;
			}
			
			atcommand "@guild \"" + .@guild_name$ + "\"";
			
			specialeffect 734;
			mes "สร้างกิลด์ ^0055FF" + .@guild_name$ + "^000000 สำเร็จเรียบร้อยแล้วค่ะ!";
			close;

		case 5: // แต่งตัวละคร (เปิดหน้าต่าง Stylist UI)
			mes "[ แต่งตัวละคร ]";
			mes "เชิญเลือกสไตล์ที่คุณชื่นชอบจากหน้าต่าง UI ได้เลยค่ะ";
			next;
			close2;
			openstylist();
			end;

		case 6: // รีเซ็ต สเตตัส (คิดค่าบริการ 1,000,000 Zeny)
			mes "[ รีเซ็ตสเตตัส ]";
			mes "การรีเซ็ตสเตตัสมีค่าใช้จ่าย ^FF77001,000,000 Zeny^000000 ค่ะ";
			mes "ต้องการดำเนินการต่อหรือไม่คะ?";
			next;
			
			if (select(" • ตกลง (รีเซ็ตสเตตัส)", " • ยกเลิก") == 2) {
				close;
			}
			
			if (Zeny < 1000000) {
				mes "[ รีเซ็ตสเตตัส ]";
				mes "^FF0000Zeny ของคุณไม่พอค่ะ!^000000 ต้องการใช้ 1,000,000 Zeny";
				close;
			}
			
			Zeny -= 1000000;
			ResetStatus;
			
			specialeffect 154;
			mes "รีเซ็ต ^0055FFสเตตัส^000000 เรียบร้อยแล้วค่ะ หักค่าบริการ 1,000,000 Zeny อัพใหม่ได้เลย!";
			close;

		case 7: // รีเซ็ต สกิล (ฟรีตามเดิม หรือปรับแก้ได้ตามต้องการ)
			ResetSkill;
			mes "รีเซ็ต ^0055FFสกิล^000000 เรียบร้อยแล้วค่ะ อัพใหม่ได้เลย!";
			close;

		case 8: // เช่าทุกอย่าง (Cart, Peco, Dragon, Falcon, Warg, Mado)
			mes "[ บริการให้เช่า ]";
			if (ismounting()) {
				mes "^FF0000กรุณาลงจากสัตว์ขี่ของท่านก่อนทำการเช่าใหม่ค่ะ^000000";
				close;
			}
			
			if (((eaclass() & EAJ_THIRDMASK) == EAJ_RANGER) && !countitem(6124)) {
				if (!checkfalcon() && getskilllv("HT_FALCON") && !checkwug()) {
					mes "คุณมีทักษะทั้ง 2 แบบ ต้องการเช่าอะไรคะ?";
					next;
					if (select(" • นกฟอลคอน (Falcon)", " • หมาป่า (Warg)") == 1) {
						setfalcon;
						mes "รับ ^0055FFนกฟอลคอน^000000 เรียบร้อยแล้วค่ะ";
					} else {
						getitem 6124, 1;
						mes "รับ ^0055FFWolf's Flute^000000 สำหรับเรียกหมาป่าเรียบร้อยแล้วค่ะ";
					}
				} else {
					getitem 6124, 1;
					mes "รับ ^0055FFWolf's Flute^000000 สำหรับเรียกหมาป่าเรียบร้อยแล้วค่ะ";
				}
			}
			else if (((eaclass() & EAJ_THIRDMASK) == EAJ_MECHANIC) && !checkcart() && getskilllv("MC_PUSHCART")) {
				if (!checkmadogear() && getskilllv("NC_MADOLICENCE")) {
					mes "คุณมีทักษะทั้ง 2 แบบ ต้องการเช่าอะไรคะ?";
					next;
					if (select(" • รถเข็น (Cart)", " • หุ่นยนต์ (Mado Gear)") == 1) {
						setcart;
						mes "รับ ^0055FFรถเข็น^000000 เรียบร้อยแล้วค่ะ";
					} else {
						setmadogear;
						mes "รับ ^0055FFหุ่นยนต์ (Mado Gear)^000000 เรียบร้อยแล้วค่ะ";
					}
				} else {
					setcart;
					mes "รับ ^0055FFรถเข็น^000000 เรียบร้อยแล้วค่ะ";
				}
			}
			else if (!checkcart() && getskilllv("MC_PUSHCART")) {
				setcart;
				mes "รับ ^0055FFรถเข็น^000000 เรียบร้อยแล้วค่ะ";
			}
			else if (!checkfalcon() && getskilllv("HT_FALCON") && !checkwug()) {
				setfalcon;
				mes "รับ ^0055FFนกฟอลคอน^000000 เรียบร้อยแล้วค่ะ";
			}
			else if (!checkriding() && !checkdragon() && getskilllv("KN_RIDING")) {
				if ((eaclass() & EAJ_THIRDMASK) == EAJ_RUNE_KNIGHT) {
					setdragon;
					mes "รับ ^0055FFมังกร (Dragon)^000000 เรียบร้อยแล้วค่ะ";
				} else {
					setriding;
					mes "รับ ^0055FFสัตว์ขี่ (Peco Peco)^000000 เรียบร้อยแล้วค่ะ";
				}
			}
			else if (!checkmadogear() && getskilllv("NC_MADOLICENCE")) {
				setmadogear;
				mes "รับ ^0055FFหุ่นยนต์ (Mado Gear)^000000 เรียบร้อยแล้วค่ะ";
			}
			else {
				mes "^FF0000เจ้ามีคุณสมบัติไม่ตรงตามข้อกำหนดในการเช่า^000000";
				mes "หรืออาจจะทำการเช่าสิ่งนั้นไปแล้วค่ะ";
				close;
			}
			
			specialeffect 154;
			close;

		case 9: // แก้บัคตัวละคร (ระบบจัดการตัวละครและไอดีอื่น)
			mes "[ แก้บัคตัวละคร ]";
			mes "สวัสดีครับ ยินดีให้บริการครับ";
			mes "เลือกเมนูที่คุณต้องการครับ";
			next;
			
			.@bug_menu = select(
				" • แก้บัคตัวละคร (ดึงกลับจุดเซฟ)",
				" • เปลี่ยน Pass",
				" • Reset การแต่งกาย",
				" • ยกเลิก"
			);
			
			switch(.@bug_menu) {
				case 1: // ใส่ชื่อตัวละครที่ต้องการ Reset
					mes "[ แก้บัคตัวละคร ]";
					mes "ใส่ชื่อตัวละครที่ต้องการ Reset";
					next;
					
					input .@target_char$;
					
					if (.@target_char$ == "") {
						mes "คุณไม่ได้ระบุชื่อตัวละครครับ";
						close;
					}
					
					warpchar "SavePoint", 0, 0, .@target_char$;
					
					specialeffect 734;
					mes "ระบบได้ทำการ Reset และส่งตัวละคร ^0055FF" + .@target_char$ + "^000000 กลับจุดเซฟเรียบร้อยแล้วครับ!";
					close;
					
				case 2: // ใส่ ID ที่ต้องการเปลี่ยน Pass
					mes "[ แก้บัคตัวละคร ]";
					mes "ใส่ ID ที่ต้องการเปลี่ยน Pass";
					next;
					
					input .@target_id$;
					
					if (.@target_id$ == "") {
						mes "คุณไม่ได้ระบุ Account ID ครับ";
						close;
					}
					
					close2;
					atcommand "@changepass " + .@target_id$;
					end;
					
				case 3: // Reset การแต่งกาย (สำหรับตัวละครอื่น)
					mes "[ แก้บัคตัวละคร ]";
					mes "ใส่ชื่อตัวละครที่ต้องการ Reset การแต่งกาย";
					next;
					
					input .@target_char$;
					
					if (.@target_char$ == "") {
						mes "คุณไม่ได้ระบุชื่อตัวละครครับ";
						close;
					}
					
					close2;
					atcommand "@changelook " + .@target_char$ + " 1 0"; 
					atcommand "@changelook " + .@target_char$ + " 2 0"; 
					atcommand "@changelook " + .@target_char$ + " 3 0"; 
					end;
					
				case 4: // ยกเลิก
					close;
			}
			close;

		case 10: // เคลือบธาตุ (Endows)
			mes "[ ระบบเคลือบธาตุอาวุธ ]";
			mes "เลือกธาตุที่ต้องการเคลือบได้เลยค่ะ (ระยะเวลา 30 นาที)";
			next;
			.@ele = select(" • ธาตุไฟ (Fire)"," • ธาตุน้ำ (Water)"," • ธาตุดิน (Earth)"," • ธาตุลม (Wind)"," • ธาตุพิษ (Poison)"," • ธาตุศักดิ์สิทธิ์ (Holy)"," • ยกเลิก");
			if (.@ele == 7) close;
			
			if (.@ele == 1) sc_start SC_FIREWEAPON, 1800000, 5;
			if (.@ele == 2) sc_start SC_WATERWEAPON, 1800000, 5;
			if (.@ele == 3) sc_start SC_EARTHWEAPON, 1800000, 5;
			if (.@ele == 4) sc_start SC_WINDWEAPON, 1800000, 5;
			if (.@ele == 5) sc_start SC_ENCHANTPOISON, 1800000, 5;
			if (.@ele == 6) sc_start SC_ASPERSIO, 1800000, 5;
			
			specialeffect 154;
			mes "เคลือบธาตุเสร็จสิ้น ขอให้สนุกกับการล่ามอนสเตอร์นะคะ!";
			close;

		case 11: // ก็อบปี้สกิล (Plagiarism)
			mes "[ ก็อบปี้สกิลสำหรับ Rogue / Shadow Chaser ]";
			if (BaseClass != Job_Rogue) {
				mes "^FF0000ฟีเจอร์นี้สงวนไว้สำหรับสายอาชีพ Rogue ขึ้นไปเท่านั้นค่ะ^000000";
				close;
			}
			mes "กรุณาใส่ ^0055FFID ของสกิล^000000 ที่คุณต้องการจำค่ะ";
			mes "(เช่น 84 = Bowling Bash, 89 = Storm Gust)";
			next;
			input .@copy_id;
			
			atcommand "@useskill " + .@copy_id + " 10 " + strcharinfo(0);
			mes "ส่งสกิล ID: " + .@copy_id + " ให้คุณจำเรียบร้อยแล้วค่ะ";
			close;

		case 12: // เพิ่มน้ำหนัก (Max Weight)
			mes "[ บริการขยายกระเป๋า ]";
			if (gympassmemory >= 10) {
				mes "น้ำหนักของคุณถูกขยายจนเต็ม ^FF0000(ระดับ 10)^000000 แล้วค่ะ";
				close;
			}
			
			if (countitem(7776) < 1) {
				mes "คุณไม่มีไอเทม ^FF0000Gym Pass^000000 ค่ะ";
				mes "จำเป็นต้องใช้ 1 ชิ้น ต่อการขยายกระเป๋า 1 ระดับนะคะ";
				close;
			}
			
			delitem 7776, 1;
			gympassmemory++;
			
			skill "ALL_INCCARRY", gympassmemory, SKILL_PERM_GRANT; 
			
			specialeffect 734;
			mes "ทำการ ^0055FFขยายกระเป๋าน้ำหนัก^000000 ขึ้นเป็นระดับ " + gympassmemory + " เรียบร้อยแล้วค่ะ!";
			
			if (gympassmemory < 10) {
				mes "เหลือโควตาขยายได้อีก ^FF0000" + (10 - gympassmemory) + "^000000 ครั้งค่ะ";
			}
			close;

		case 13: // สัตว์เลี้ยงสนิททันที
			if (getpetinfo(PETINFO_ID) == 0) {
				mes "^FF0000คุณยังไม่ได้ฟักสัตว์เลี้ยงเลยค่ะ^000000";
				close;
			}
			
			atcommand "@petfriendly 1000"; 
			
			specialeffect 338;
			mes "สัตว์เลี้ยงของคุณรู้สึก ^0055FFสนิทสนม^000000 กับคุณเต็มที่แล้วค่ะ!";
			close;

		case 14: // โฮมุนครุส (Homunculus)
			mes "[ ระบบจัดการ Homunculus ]";
			if (BaseClass != Job_Alchemist) {
				mes "^FF0000ฟีเจอร์นี้สำหรับอาชีพสาย Alchemist เท่านั้นค่ะ^000000";
				close;
			}
			.@hom = select(" • เรียกโฮมุนครุส (Call)", " • ชุบชีวิต (Resurrect)", " • สนิทสนมทันที (Intimacy Max)", " • ยกเลิก");
			if (.@hom == 4) close;

			if (.@hom == 1) { skill 247, 1, 0; }
			if (.@hom == 2) { skill 243, 1, 0; }
			if (.@hom == 3) { 
				if (gethominfo(0) > 0) {
					atcommand "@homintimacy 100000";
					mes "โฮมุนครุสของคุณสนิทสนมถึงขีดสุดแล้วค่ะ!";
				} else {
					mes "^FF0000คุณยังไม่มีโฮมุนครุส หรือยังไม่ได้เรียกออกมาค่ะ^000000";
				}
			}
			close;

		case 15: // เอาไว้คราวหน้า !!
			mes "ขอบคุณที่ใช้บริการ ^0055FFKafra Corporation^000000 ค่ะ";
			mes "เดินทางปลอดภัยนะคะ!";
			close;
	}
