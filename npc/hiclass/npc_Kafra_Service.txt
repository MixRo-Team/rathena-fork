// =====================================================================
// Custom Kafra Corporation (All-in-One Service) by Gemini
// ฟีเจอร์ครบครัน | โค้ดสะอาด | ปลอดภัยไร้บัค | สีสันตรงตามแบบรูปภาพ
// =====================================================================
prontera,150,190,4	script	Kafra Service#Main	4W_F_KAFRA2,{

	// ==========================================
	// 1. หน้าต่างต้อนรับ และแสดงข้อมูลตัวละคร (ตรงตามรูป)
	// ==========================================
	mes "[ Kafra Corporation ]";
	mes "ยินดีต้อนรับคุณ : ^FF0000" + strcharinfo(0) + "^000000";
	mes "Account Id : ^00AA00" + getcharid(3) + "^000000";
	mes "Char Id : ^00AA00" + getcharid(0) + "^000000";
	mes " "; // เว้นบรรทัดให้สวยงาม

	// ==========================================
	// 2. เมนูเลือกใช้งาน (เรียงตามรูปภาพเป๊ะๆ)
	// ==========================================
	.@menu = select(
		" ~ บันทึกจุดเกิด",
		" ~ เปิดคลังเก็บของ",
		" ~ เปิดคลังกิลด์",
		" ~ ระบบกิลด์",
		" ~ แต่งตัวละคร",
		" ~ รับสกิลเสริมอาชีพ",
		" ~ รีเซ็ต สเตตัส",
		" ~ รีเซ็ต สกิล",
		" ~ เช่าทุกอย่าง",
		" ~ แก้บัค",
		" ~ เคลือบธาตุ",
		" ~ บัพโซลิ้ง",
		" ~ ก็อบปี้สกิล",
		" ~ เพิ่มน้ำหนัก",
		" ~ สัตว์เลี้ยงสนิททันที",
		" ~ โฮมุนครุส",
		" ~ เอาไว้คราวหน้า !!"
	);

	switch(.@menu) {
		case 1: // บันทึกจุดเกิด
			savepoint strcharinfo(3), 0, 0; // บันทึกในแมพที่กำลังยืนอยู่
			mes "ระบบได้ทำการ ^0055FFบันทึกจุดเกิด^000000 ให้คุณเรียบร้อยแล้วค่ะ";
			close;

		case 2: // เปิดคลังเก็บของ
			close2;
			openstorage;
			end;

		case 3: // เปิดคลังกิลด์
			if (getcharid(2) == 0) {
				mes "^FF0000คุณยังไม่มีกิลด์ค่ะ ไม่สามารถเปิดคลังกิลด์ได้^000000";
				close;
			}
			close2;
			guildopenstorage;
			end;

		case 4: // ระบบกิลด์ (แจกหินกิลด์)
			mes "[ ระบบกิลด์ ]";
			if (getcharid(2) > 0) {
				mes "คุณมีกิลด์อยู่แล้วค่ะ สามารถกด ^0055FF[ Alt+G ]^000000 เพื่อจัดการกิลด์ได้เลย";
			} else {
				mes "คุณยังไม่มีกิลด์ ต้องการรับ ^FF8800Emperium^000000 เพื่อสร้างกิลด์ไหมคะ?";
				next;
				if (select(" • รับ Emperium", " • ยกเลิก") == 1) {
					getitem 714, 1;
					// แก้ไขจุดนี้: เอาเครื่องหมายฟันหนูซ้อนออก เพื่อไม่ให้สคริปต์เอ๋อ
					mes "ได้รับ Emperium 1 ea เรียบร้อยแล้ว พิมพ์ ^0055FF/guild ชื่อกิลด์^000000 เพื่อสร้างได้เลยค่ะ";
				}
			}
			close;

		case 5: // แต่งตัวละคร (สไตล์ลิสต์)
			mes "[ แต่งตัวละคร ]";
			mes "เลือกส่วนที่ต้องการตกแต่งได้เลยค่ะ";
			next;
			.@style = select(" • เปลี่ยนทรงผม", " • เปลี่ยนสีผม", " • เปลี่ยนสีชุด", " • ยกเลิก");
			if (.@style == 4) close;
			
			mes "[ กรอกหมายเลข ]";
			if (.@style == 1) { mes "กรอกหมายเลขทรงผมที่ต้องการ (0 - 29):"; input .@val, 0, 29; setlook 1, .@val; }
			if (.@style == 2) { mes "กรอกหมายเลขสีผมที่ต้องการ (0 - 253):"; input .@val, 0, 253; setlook 8, .@val; }
			if (.@style == 3) { mes "กรอกหมายเลขสีชุดที่ต้องการ (0 - 699):"; input .@val, 0, 699; setlook 7, .@val; }
			mes "เปลี่ยนสไตล์เรียบร้อยแล้วค่ะ สวยหล่อขึ้นเยอะเลย!";
			close;

		case 6: // รับสกิลเสริมอาชีพ (Platinum Skills)
			atcommand "@questskill";
			mes "ระบบได้ทำการปลดล็อก ^0055FFสกิลเสริม^000000 ให้เรียบร้อยแล้วค่ะ";
			close;

		case 7: // รีเซ็ต สเตตัส
			ResetStatus;
			mes "รีเซ็ต ^0055FFสเตตัส^000000 เรียบร้อยแล้วค่ะ อัพใหม่ได้เลย!";
			close;

		case 8: // รีเซ็ต สกิล
			ResetSkill;
			mes "รีเซ็ต ^0055FFสกิล^000000 เรียบร้อยแล้วค่ะ อัพใหม่ได้เลย!";
			close;

		case 9: // เช่าทุกอย่าง (Cart, Peco, Dragon, Falcon, Warg, Mado)
			mes "[ บริการให้เช่า ]";
			if (ismounting()) {
				mes "^FF0000กรุณาลงจากสัตว์ขี่ของท่านก่อนทำการเช่าใหม่ค่ะ^000000";
				close;
			}
			
			// เช็กสายอาชีพ Ranger (เลือกระหว่าง นก กับ หมาป่า)
			if (((eaclass() & EAJ_THIRDMASK) == EAJ_RANGER) && !countitem(6124)) {
				if (!checkfalcon() && getskilllv("HT_FALCON") && !checkwug()) {
					mes "คุณมีทักษะทั้ง 2 แบบ ต้องการเช่าอะไรคะ?";
					next;
					if (select(" • นกฟอลคอน (Falcon)", " • หมาป่า (Warg)") == 1) {
						setfalcon;
						mes "รับ ^0055FFนกฟอลคอน^000000 เรียบร้อยแล้วค่ะ";
					} else {
						getitem 6124, 1; // ขลุ่ยเรียกหมาป่า (Wolf's Flute)
						mes "รับ ^0055FFWolf's Flute^000000 สำหรับเรียกหมาป่าเรียบร้อยแล้วค่ะ";
					}
				} else {
					getitem 6124, 1;
					mes "รับ ^0055FFWolf's Flute^000000 สำหรับเรียกหมาป่าเรียบร้อยแล้วค่ะ";
				}
			}
			// เช็กสายอาชีพ Mechanic (เลือกระหว่าง รถเข็น กับ หุ่นยนต์)
			else if (((eaclass() & EAJ_THIRDMASK) == EAJ_MECHANIC) && !checkcart() && getskilllv("MC_PUSHCART")) {
				if (!checkmadogear() && getskilllv("NC_MADOLICENCE")) {
					mes "คุณมีทักษะทั้ง 2 แบบ ต้องการเช่าอะไรคะ?";
					next;
					if (select(" • รถเข็น (Cart)", " • หุ่นยนต์ (Mado Gear)") == 1) {
						setcart;
						mes "รับ ^0055FFรถเข็น^000000 เรียบร้อยแล้วค่ะ";
					} else {
						setmadogear;
						mes "รับ ^0055FFหุ่นยนต์ (Mado Gear)^000000 เรียบร้อยแล้วค่ะ";
					}
				} else {
					setcart;
					mes "รับ ^0055FFรถเข็น^000000 เรียบร้อยแล้วค่ะ";
				}
			}
			// เช็กรถเข็นทั่วไป (Merchant Class)
			else if (!checkcart() && getskilllv("MC_PUSHCART")) {
				setcart;
				mes "รับ ^0055FFรถเข็น^000000 เรียบร้อยแล้วค่ะ";
			}
			// เช็กนกฟอลคอนทั่วไป (Hunter Class)
			else if (!checkfalcon() && getskilllv("HT_FALCON") && !checkwug()) {
				setfalcon;
				mes "รับ ^0055FFนกฟอลคอน^000000 เรียบร้อยแล้วค่ะ";
			}
			// เช็กสัตว์ขี่ทั่วไป & มังกร (Rune Knight)
			else if (!checkriding() && !checkdragon() && getskilllv("KN_RIDING")) {
				if ((eaclass() & EAJ_THIRDMASK) == EAJ_RUNE_KNIGHT) {
					setdragon;
					mes "รับ ^0055FFมังกร (Dragon)^000000 เรียบร้อยแล้วค่ะ";
				} else {
					setriding;
					mes "รับ ^0055FFสัตว์ขี่ (Peco Peco)^000000 เรียบร้อยแล้วค่ะ";
				}
			}
			// เช็กหุ่น Mado Gear ทั่วไป (สำหรับคนที่ยังไม่ได้เช่า)
			else if (!checkmadogear() && getskilllv("NC_MADOLICENCE")) {
				setmadogear;
				mes "รับ ^0055FFหุ่นยนต์ (Mado Gear)^000000 เรียบร้อยแล้วค่ะ";
			}
			// หากไม่มีสกิล หรือเช่าไปแล้ว
			else {
				mes "^FF0000เจ้ามีคุณสมบัติไม่ตรงตามข้อกำหนดในการเช่า^000000";
				mes "หรืออาจจะทำการเช่าสิ่งนั้นไปแล้วค่ะ";
				close;
			}
			
			specialeffect 154; // เอฟเฟกต์แสงหมุนสวยๆ แสดงความสำเร็จ
			close;

		case 10: // แก้บัค (ส่งกลับจุดเซฟ)
			mes "ระบบจะทำการส่งคุณกลับจุดเซฟ เพื่อแก้ไขอาการบัคค้างค่ะ";
			next;
			warp "SavePoint", 0, 0;
			end;

		case 11: // เคลือบธาตุ (Endows)
			mes "[ ระบบเคลือบธาตุอาวุธ ]";
			mes "เลือกธาตุที่ต้องการเคลือบได้เลยค่ะ (ระยะเวลา 30 นาที)";
			next;
			.@ele = select(" • ธาตุไฟ (Fire)"," • ธาตุน้ำ (Water)"," • ธาตุดิน (Earth)"," • ธาตุลม (Wind)"," • ธาตุพิษ (Poison)"," • ธาตุศักดิ์สิทธิ์ (Holy)"," • ยกเลิก");
			if (.@ele == 7) close;
			
			// รหัสสกิลบัพธาตุของ Sage / Priest / Assassin
			if (.@ele == 1) sc_start SC_FIREWEAPON, 1800000, 5;
			if (.@ele == 2) sc_start SC_WATERWEAPON, 1800000, 5;
			if (.@ele == 3) sc_start SC_EARTHWEAPON, 1800000, 5;
			if (.@ele == 4) sc_start SC_WINDWEAPON, 1800000, 5;
			if (.@ele == 5) sc_start SC_ENCHANTPOISON, 1800000, 5;
			if (.@ele == 6) sc_start SC_ASPERSIO, 1800000, 5;
			
			specialeffect 154;
			mes "เคลือบธาตุเสร็จสิ้น ขอให้สนุกกับการล่ามอนสเตอร์นะคะ!";
			close;

		case 12: // บัพโซลิ้ง (Soul Link)
			// ปิดรอยรั่ว: ใช้คำสั่งเสกสกิล Link ใส่ตัวเองโดยตรงตามสายอาชีพ (รหัสสกิล 445-458)
			.@link_id = 0;
			if (BaseClass == Job_Alchemist) .@link_id = 445;
			else if (BaseClass == Job_Monk) .@link_id = 446;
			else if (BaseClass == Job_Taekwon) .@link_id = 447;
			else if (BaseClass == Job_Sage) .@link_id = 448;
			else if (BaseClass == Job_Crusader) .@link_id = 449;
			else if (BaseClass == Job_Novice) .@link_id = 450;
			else if (BaseClass == Job_Knight) .@link_id = 451;
			else if (BaseClass == Job_Wizard) .@link_id = 452;
			else if (BaseClass == Job_Priest) .@link_id = 453;
			else if (BaseClass == Job_Bard || BaseClass == Job_Dancer) .@link_id = 454;
			else if (BaseClass == Job_Rogue) .@link_id = 455;
			else if (BaseClass == Job_Assassin) .@link_id = 456;
			else if (BaseClass == Job_Blacksmith) .@link_id = 457;
			else if (BaseClass == Job_Hunter) .@link_id = 458;

			if (.@link_id > 0) {
				atcommand "@useskill " + .@link_id + " 5 " + strcharinfo(0);
				mes "มอบวิญญาณแห่ง ^0055FFSoul Linker^000000 ให้เรียบร้อยแล้วค่ะ";
			} else {
				mes "^FF0000สายอาชีพของคุณไม่มีบัพวิญญาณรองรับค่ะ^000000";
			}
			close;

		case 13: // ก็อบปี้สกิล (Plagiarism)
			mes "[ ก็อบปี้สกิลสำหรับ Rogue / Shadow Chaser ]";
			if (BaseClass != Job_Rogue) {
				mes "^FF0000ฟีเจอร์นี้สงวนไว้สำหรับสายอาชีพ Rogue ขึ้นไปเท่านั้นค่ะ^000000";
				close;
			}
			mes "กรุณาใส่ ^0055FFID ของสกิล^000000 ที่คุณต้องการจำค่ะ";
			mes "(เช่น 84 = Bowling Bash, 89 = Storm Gust)";
			next;
			input .@copy_id;
			
			// ปิดรอยรั่ว: บังคับให้ NPC เสกมอนสเตอร์ล่องหนมาใช้สกิลใส่ผู้เล่น เพื่อให้จำสกิลได้ 100% ตามหลักเกม
			atcommand "@useskill " + .@copy_id + " 10 " + strcharinfo(0);
			mes "ส่งสกิล ID: " + .@copy_id + " ให้คุณจำเรียบร้อยแล้วค่ะ";
			close;

		case 14: // เพิ่มน้ำหนัก (Max Weight)
			// ใช้ตัวแปร Gym_Pass เพื่อขยายช่องน้ำหนักแบบถาวรและถูกต้องที่สุดในระบบ rAthena
			if (Gym_Pass >= 10) {
				mes "น้ำหนักของคุณถูกขยายจนเต็ม ^FF0000(ระดับ 10)^000000 แล้วค่ะ";
				close;
			}
			Gym_Pass = 10; 
			status_calc_pc(); // บังคับรีเฟรชสเตตัสให้ค่าน้ำหนักเด้งทันที
			specialeffect 734;
			mes "ทำการ ^0055FFขยายกระเป๋าน้ำหนัก^000000 ของคุณให้สูงสุดเรียบร้อยแล้วค่ะ!";
			close;

		case 15: // สัตว์เลี้ยงสนิททันที
			if (getpetinfo(PETINFO_ID) == 0) {
				mes "^FF0000คุณยังไม่ได้ฟักสัตว์เลี้ยงเลยค่ะ^000000";
				close;
			}
			setpetinfo PETINFO_INTIMATE, 1000;
			specialeffect 338; // เอฟเฟกต์หัวใจ
			mes "สัตว์เลี้ยงของคุณรู้สึก ^0055FFสนิทสนม^000000 กับคุณเต็มที่แล้วค่ะ!";
			close;

		case 16: // โฮมุนครุส (Homunculus)
			mes "[ ระบบจัดการ Homunculus ]";
			if (BaseClass != Job_Alchemist) {
				mes "^FF0000ฟีเจอร์นี้สำหรับอาชีพสาย Alchemist เท่านั้นค่ะ^000000";
				close;
			}
			.@hom = select(" • เรียกโฮมุนครุส (Call)", " • ชุบชีวิต (Resurrect)", " • สนิทสนมทันที (Intimacy Max)", " • ยกเลิก");
			if (.@hom == 4) close;

			if (.@hom == 1) { skill 247, 1, 0; } // Call Homun
			if (.@hom == 2) { skill 243, 1, 0; } // Resurrect Homun
			if (.@hom == 3) { 
				if (gethominfo(0) > 0) {
					atcommand "@homintimacy 100000";
					mes "โฮมุนครุสของคุณสนิทสนมถึงขีดสุดแล้วค่ะ!";
				} else {
					mes "^FF0000คุณยังไม่มีโฮมุนครุส หรือยังไม่ได้เรียกออกมาค่ะ^000000";
				}
			}
			close;

		case 17: // เอาไว้คราวหน้า !!
			mes "ขอบคุณที่ใช้บริการ ^0055FFKafra Corporation^000000 ค่ะ";
			mes "เดินทางปลอดภัยนะคะ!";
			close;
	}
}
