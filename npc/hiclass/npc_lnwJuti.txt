prontera,162,194,4	script	เทพเจ้าจุติ	4_M_FAIRYAVANT,{
	// ==========================================
	// --- ตั้งค่าระบบจุติหลัก ---
	// ==========================================
	.@MaxLevel = 150;      // เลเวลตันที่ต้องการให้จุติ
	.@FreeLimit = 50;      // จำนวนครั้งที่จุติฟรี
	.@BonusPoint = 500;    // โบนัส Status Point ที่จะได้รับต่อการจุติ 1 รอบ
	
	// --- ตั้งค่าทรัพยากรเมื่อหมดโควต้าฟรี (ครั้งที่ 50 เป็นต้นไป) ---
	.@CostZeny = 1000000;  // จำนวนเงิน Zeny
	.@CostItem = 61502;      // ID ไอเทม (เปลี่ยนเลข 61502 เป็น ID เหรียญของเซิร์ฟจารได้เลย)
	.@CostAmount = 20;     // จำนวนไอเทมที่ใช้

L_Main:
	mes "[ เทพเจ้าจุติ ]";
	mes "ท่านจุติไปแล้วทั้งหมด ^FF0000" + JutiCount + "^000000 ครั้ง";
	mes "---";
	
	// แสดงเงื่อนไขการทำงานให้ผู้เล่นเข้าใจง่าย
	mes "^0055FF[ เงื่อนไขการจุติ ]^000000";
	mes "- Base Level จะกลับไปเป็น 1";
	mes "- ^bb0000สเตตัสเดิมจะไม่ถูกรีเซ็ต^000000";
	mes "- ได้รับโบนัส Status Point ^00bb00" + .@BonusPoint + " แต้ม^000000";
	mes "---";
	
	// ตรวจสอบและแสดงโปรโมชั่นหรือค่าธรรมเนียม
	if (JutiCount < .@FreeLimit) {
		mes "^00FF00** โปรโมชั่น: จุติฟรี (เหลืออีก " + (.@FreeLimit - JutiCount) + " ครั้ง) **^000000";
	} else {
		mes "ค่าธรรมเนียมการจุติ:";
		mes "- เงิน ^0000FF1,000,000^000000 Zeny";
		mes "- ไอเทม " + getitemname(.@CostItem) + " จำนวน ^FF0000" + .@CostAmount + "^000000 ชิ้น";
	}
	next;

	switch(select(" ~ จุติเลเวล: ~ เปิดระบบจุติเลเวลออโต้: ~ ปิดระบบจุติเลเวลออโต้")) {
		
		case 1: // ================= เมนู จุติเลเวล (กดเอง) =================
			if (BaseLevel < .@MaxLevel) {
				mes "[ เทพเจ้าจุติ ]";
				mes "เลเวลของท่านยังไม่ถึง " + .@MaxLevel + " ครับ";
				close;
			}

			mes "[ เทพเจ้าจุติ ]";
			if (JutiCount < .@FreeLimit) {
				mes "ท่านอยู่ในช่วงจุติฟรี ต้องการจุติเลยหรือไม่?";
			} else {
				mes "ต้องใช้เงิน ^0000FF1,000,000 Zeny^000000";
				mes "และไอเทม " + getitemname(.@CostItem) + " อีก ^FF0000" + .@CostAmount + "^000000 ชิ้น";
				mes "ต้องการจุติเลยหรือไม่?";
			}
			next;
			
			if(select(" ~ ตกลง: ~ ยกเลิก") == 2) {
				close;
			}

			// หักทรัพยากรกรณีเกินโควต้าฟรี (เช็คทั้ง Zeny และ Item)
			if (JutiCount >= .@FreeLimit) {
				if (Zeny < .@CostZeny || countitem(.@CostItem) < .@CostAmount) {
					mes "[ เทพเจ้าจุติ ]";
					mes "ทรัพยากรของท่านไม่เพียงพอครับ!";
					close;
				}
				Zeny -= .@CostZeny;
				delitem .@CostItem, .@CostAmount; 
			}

			// เริ่มกระบวนการจุติ
			resetlvl(1);                  // รีเซ็ต Base Level ไปที่ 1
			StatusPoint += .@BonusPoint;  // แจกแต้มโบนัสให้ตัวละคร
			JutiCount++;                  // บวกจำนวนรอบการจุติเพิ่ม 1
			specialeffect2 EF_ANGEL2;     // แสดงเอฟเฟกต์เทวดาตอนเปลี่ยนคลาส
			
			mes "[ เทพเจ้าจุติ ]";
			mes "จุติสำเร็จ! ท่านได้รับ Status Point เพิ่ม ^00bb00" + .@BonusPoint + " แต้ม^000000";
			close;

		case 2: // ================= เปิดระบบออโต้ =================
			AutoJuti = 1;
			mes "[ เทพเจ้าจุติ ]";
			mes "เปิดระบบจุติเลเวลออโต้แล้ว";
			mes "*(ฟรี " + .@FreeLimit + " ครั้งแรก";
			mes "หลังจากนั้นจะหักทรัพยากรปกติ)*";
			close;

		case 3: // ================= ปิดระบบออโต้ =================
			AutoJuti = 0;
			mes "[ เทพเจ้าจุติ ]";
			mes "ปิดระบบจุติเลเวลออโต้แล้ว";
			close;
	}
}

// ==========================================
// --- ระบบเบื้องหลังการจุติอัตโนมัติ (Auto Juti) ---
// ==========================================
-	script	Auto_Juti_System	-1,{
OnPCBaseUpEvent:
	// ตรวจสอบสถานะการเปิดออโต้
	if (AutoJuti == 0) end;

	// ต้องตั้งค่าตรงนี้ให้ตรงกับ NPC หลัก
	.@MaxLevel = 150;
	.@FreeLimit = 50;
	.@BonusPoint = 500;
	.@CostZeny = 1000000;
	.@CostItem = 61502;      // แก้ ID ให้ตรงกับ NPC หลัก
	.@CostAmount = 20;

	// เงื่อนไข: เลเวลยังไม่ถึงจุดจุติ ให้ข้ามไป
	if (BaseLevel < .@MaxLevel) end;

	// เงื่อนไข: ถ้ารอบจุติเกินโควต้าฟรี ให้หักทรัพยากร
	if (JutiCount >= .@FreeLimit) {
		if (Zeny < .@CostZeny || countitem(.@CostItem) < .@CostAmount) {
			dispbottom "[เทพเจ้าจุติ] ทรัพยากรไม่พอสำหรับการจุติ ระบบได้ทำการปิดออโต้ชั่วคราว!";
			AutoJuti = 0; 
			end;
		}
		Zeny -= .@CostZeny;
		delitem .@CostItem, .@CostAmount;
	}

	// เริ่มกระบวนการจุติออโต้ (ไม่รีสเตตัส ไม่รีสกิล)
	resetlvl(1);
	StatusPoint += .@BonusPoint; 
	JutiCount++;
	specialeffect2 EF_ANGEL2;
	
	// แจ้งเตือนข้อความให้ผู้เล่นเห็นในช่องแชท
	dispbottom "[เทพเจ้าจุติ] จุติอัตโนมัติสำเร็จ! (รอบที่ " + JutiCount + " | ได้โบนัส " + .@BonusPoint + " แต้ม)";
	end;
}
